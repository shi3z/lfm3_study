Supports:
- integers and floats
- operators: + - * / ^
- parentheses
- unary minus (e.g. -3, -(2+3))

            i = j

        else:

def evaluate(s: str) -> float:
    if s is None or s.strip() == '':
        raise ValueError("empty expression")
    tokens = tokenize(s)
    pos = 0

    def peek():
        return tokens[pos] if pos < len(tokens) else None

    def consume():
        nonlocal pos
        token = tokens[pos]
        pos += 1
        return token

    def parse_number():
        nonlocal pos
        if pos >= len(tokens):
            raise ValueError("expected number")
        num_str = tokens[pos]
        if num_str == '':
            raise ValueError("empty number token")
        val = float(num_str)
        pos += 1
        return val

    def parse_base():
        nonlocal pos
        if pos >= len(tokens):
            raise ValueError("expected base")
        token = tokens[pos]
        if token == '(':
            consume()
            node = parse_expr()
            if pos >= len(tokens) or tokens[pos] != ')':
